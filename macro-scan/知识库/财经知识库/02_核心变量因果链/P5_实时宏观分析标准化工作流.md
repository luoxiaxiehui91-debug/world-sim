### 3.1 RAG 检索（知识库增强）

```python
import chromadb
from chromadb.config import Settings

CHROMA_DIR = "RAG/chroma_db"
COLLECTION = "langchain"

def rag_query(query, n_results=5):
    """从知识库检索相关上下文"""
    client = chromadb.PersistentClient(path=CHROMA_DIR)
    collection = client.get_collection(name=COLLECTION)
    
    # 使用 nomic-embed-text 编码查询
    results = collection.query(
        query_texts=[query],
        n_results=n_results
    )
    
    chunks = results["documents"][0]
    metadatas = results["metadatas"][0]
    return list(zip(chunks, metadatas))
```

### 3.3 衰退/通胀信号评分

```python
def score_recession_risk(indicators):
    """
    衰退风险评分（0-100）
    满足任意3项 → 衰退概率显著上升
    """
    signals = []
    
    # 1. 利差倒挂
    t10y2y = indicators.get("T10Y2Y", {}).get("value", 0)
    if t10y2y < -50:  # -50bp以下
        signals.append(("利差深度倒挂", t10y2y))
    
    # 2. 初请上升
    icsa = indicators.get("ICSA", {}).get("value", 0)
    if icsa > 25:
        signals.append(("初请高位", icsa))
    
    # 3. 失业率上升
    unrate = indicators.get("UNRATE", {}).get("value", 0)
    if unrate > 5:
        signals.append(("失业率上升", unrate))
    
    # 4. 消费信心低位
    # ... (UMCSENT)
    
    # 5. M2收缩
    # ... (M2SL同比)
    
    n = len(signals)
    risk_map = {
        0: ("极低", 5),
        1: ("低", 15),
        2: ("中等", 30),
        3: ("较高", 50),
        4: ("高", 70),
        5: ("严重", 85),
    }
    
    label, score = risk_map.get(n, ("极端", 95))
    return label, score, signals

def score_inflation_risk(indicators):
    """
    通胀风险评分（0-100）
    """
    signals = []
    
    # CPI同比
    cpi = indicators.get("CPIAUCSL", {}).get("value", 0)
    if cpi > 5:
        signals.append(("CPI高位", cpi))
    elif cpi > 3:
        signals.append(("CPI偏高", cpi))
    
    # 核心PCE
    pce = indicators.get("PCECTPI", {}).get("value", 0)
    if pce > 3:
        signals.append(("核心PCE偏高", pce))
    
    # 油价
    # ... (DCOILWTICO)
    
    # M2
    # ... (M2SL同比)
    
    n = len(signals)
    risk_map = {
        0: ("极低", 5),
        1: ("低", 15),
        2: ("中等", 35),
        3: ("较高", 55),
        4: ("高", 75),
        5: ("严重", 90),
    }
    
    label, score = risk_map.get(n, ("极端", 98))
    return label, score, signals
```

## 四、推理输出层（Layer 3）

### 4.1 Ollama 分析提示词模板

```python
ANALYSIS_PROMPT_TEMPLATE = """你是一位宏观经济学家，请基于以下信息，撰写一份结构化分析报告。

## 当前宏观数据（{date}）
{current_data_table}

## 衰退风险评估
风险等级：{recession_label}（{recession_score}/100）
触发信号：
{recession_signals}

## 通胀风险评估  
风险等级：{inflation_label}（{inflation_score}/100）
触发信号：
{inflation_signals}

## 历史情景对照
最匹配的危机情景：
{crisis_matches}

## 知识库相关背景（RAG检索）
{rag_context}

## 分析要求：
1. **核心判断**：当前宏观环境处于什么阶段？扩张/放缓/衰退/滞胀？
2. **主要风险**：1-3个最重要的风险因素
3. **情景分析**：最可能的3种情景（基准/乐观/悲观），含概率
4. **政策响应**：美联储当前立场的适当性
5. **大类资产影响**：股票/债券/大宗商品/汇率的方向判断
6. **领先指标追踪**：未来1-3个月需重点关注哪些指标

请用简洁的中文撰写，总字数控制在1500字以内。
"""

def generate_report(indicators, rag_context, crisis_matches, recession_result, inflation_result):
    """生成完整分析报告"""
    
    # 组装数据表格
    data_lines = []
    for sid, info in indicators.items():
        if info["value"] is not None:
            data_lines.append(f"- {info['name']}：{info['value']}（{info['date']}）")
    data_table = "\n".join(data_lines)
    
    # 组装信号
    r_label, r_score, r_signals = recession_result
    i_label, i_score, i_signals = inflation_result
    
    recession_signals = "\n".join([f"  - {s[0]}：{s[1]}" for s in r_signals]) or "  无明显衰退信号"
    inflation_signals = "\n".join([f"  - {s[0]}：{s[1]}" for s in i_signals]) or "  无明显通胀压力"
    
    # 历史情景
    crisis_text = "\n".join([f"  {i+1}. {c[0]}（匹配分：{c[1]}）" for i, c in enumerate(crisis_matches)])
    
    # RAG上下文
    rag_text = "\n\n".join([f"[{i+1}] {chunk[:300]}..." for i, chunk in enumerate(rag_context)])
    
    prompt = ANALYSIS_PROMPT_TEMPLATE.format(
        date=datetime.now().strftime("%Y-%m-%d"),
        current_data_table=data_table,
        recession_label=r_label,
        recession_score=r_score,
        recession_signals=recession_signals,
        inflation_label=i_label,
        inflation_score=i_score,
        inflation_signals=inflation_signals,
        crisis_matches=crisis_text,
        rag_context=rag_text,
    )
    
    return prompt
```

### 4.2 Ollama 调用

```python
import requests

OLLAMA_URL = "http://localhost:11434/api/generate"
MODEL = "qwen3-vl:8b"

def call_ollama(prompt, stream=False):
    """调用Ollama生成分析"""
    payload = {
        "model": MODEL,
        "prompt": prompt,
        "stream": stream,
        "options": {
            "temperature": 0.3,  # 较低温度，保证一致性
            "num_predict": 2048,  # 控制输出长度
        }
    }
    
    resp = requests.post(OLLAMA_URL, json=payload, timeout=60)
    return resp.json()["response"]
```

## 五、标准化分析流程（完整脚本）

```python
def run_macro_analysis(topic="综合", depth="standard"):
    """
    主分析流程
    
    Args:
        topic: "综合" | "衰退" | "通胀" | "市场" | "地缘"
        depth: "quick"(5分钟) | "standard"(15分钟) | "deep"(30分钟)
    """
    
    print("=" * 60)
    print(f"宏观分析：{topic} | 深度：{depth}")
    print("=" * 60)
    
    # Step 1: 获取当前数据
    print("[1/5] 获取实时指标...")
    indicators = get_current_snapshot()
    
    # Step 2: 衰退/通胀评分
    print("[2/5] 计算风险评分...")
    recession = score_recession_risk(indicators)
    inflation = score_inflation_risk(indicators)
    
    # Step 3: 历史情景匹配
    print("[3/5] 匹配历史情景...")
    crises = match_crisis(indicators)
    
    # Step 4: RAG检索
    print("[4/5] 检索知识库...")
    query_map = {
        "综合": "宏观经济分析框架衰退通胀资产配置",
        "衰退": "衰退风险利差倒挂失业率上升信贷收缩",
        "通胀": "CPI通胀原油货币供应M2工资传导",
        "市场": "股市暴跌流动性枯竭信用利差扩大",
        "地缘": "地缘冲突能源供应供应链大宗商品",
    }
    rag_chunks = rag_query(query_map.get(topic, query_map["综合"]), n_results=3)
    
    # Step 5: 生成报告
    print("[5/5] 生成分析报告...")
    prompt = generate_report(
        indicators=indicators,
        rag_context=[c[0] for c in rag_chunks],
        crisis_matches=crises,
        recession_result=recession,
        inflation_result=inflation,
    )
    
    report = call_ollama(prompt)
    
    return {
        "report": report,
        "indicators": indicators,
        "recession": recession,
        "inflation": inflation,
        "crises": crises,
    }
```

## 七、质量控制检查清单

每次分析输出前，检查以下要素：

- [ ] 数据时效性：指标日期是否在合理范围内？（当前季度或上月）
- [ ] 异常值处理：是否有指标超出历史3σ范围？（需特殊说明）
- [ ] 信号一致性：多个指标方向是否一致？（不一致时需权衡）
- [ ] 历史对照：是否有历史情景与当前高度相似？
- [ ] 政策协调：分析是否考虑了美联储/ECB/人民银行的最新表态？
- [ ] 尾部风险：是否考虑了低概率高影响的尾部风险？
